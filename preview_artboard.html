📄 distorsion_eblouissement.gdshader

shader_type canvas_item; // Distorsion & Bloom d'éblouissement (V5.3) — Candela 2D // Aberration chromatique radiale et ondulation thermique d'objectif ébloui. uniform sampler2D screen_texture : hint_screen_texture, filter_linear_mipmap; uniform float niveau_eblouissement : hint_range(0.0, 1.0) = 0.0; uniform float aberration_chromatique : hint_range(0.0, 0.05) = 0.015; uniform float force_ondulation : hint_range(0.0, 0.05) = 0.008; void fragment() { vec2 uv = SCREEN_UV; vec2 center_offset = uv - vec2(0.5); float dist_center = length(center_offset); // Ondulation thermique / réfraction d'air chaud float wave = sin(uv.y * 40.0 + TIME * 8.0) * cos(uv.x * 30.0 + TIME * 6.0); vec2 distorted_uv = uv + center_offset * wave * force_ondulation * niveau_eblouissement; // Séparation des canaux RGB (aberration chromatique radiale) vec2 shift = center_offset * aberration_chromatique * niveau_eblouissement * (dist_center + 0.2); float r = texture(screen_texture, distorted_uv + shift).r; float g = texture(screen_texture, distorted_uv).g; float b = texture(screen_texture, distorted_uv - shift).b; vec4 col = vec4(r, g, b, 1.0); // Voile chaud additif sur forte exposition vec4 bloom_chaud = vec4(0.98, 0.91, 0.80, 1.0) * pow(niveau_eblouissement, 1.5) * 0.4; COLOR = col + bloom_chaud; }

📄 poussiere_faisceau.gdshader

shader_type canvas_item; render_mode blend_add; // Poussière dans le faisceau (V5.5) — Candela 2D // Particules minuscules, sans lumière propre, révélées par la lumière directe. uniform float intensite_base : hint_range(0.0, 1.0) = 0.45; uniform float vitesse_derivation : hint_range(0.0, 2.0) = 0.35; uniform float taille_grain : hint_range(0.5, 5.0) = 1.2; uniform vec4 teinte_halogene : source_color = vec4(0.98, 0.91, 0.80, 1.0); float hash21(vec2 p) { p = fract(p * vec2(234.34, 435.345)); p += dot(p, p + 34.23); return fract(p.x * p.y); } void fragment() { vec2 uv = UV; float t = TIME * vitesse_derivation; // Grille de poussières flottantes vec2 grid = uv * 30.0 * taille_grain; vec2 id = floor(grid); vec2 gv = fract(grid) - 0.5; float h = hash21(id); // Mouvement brownien lent vec2 offset = vec2( sin(t * 1.3 + h * 6.28), cos(t * 0.9 + h * 6.28) ) * 0.25; float d = length(gv - offset); // Particule douce float spark = smoothstep(0.18, 0.0, d); // Scintillement d'interférence float flicker = sin(t * 2.5 + h * 12.56) * 0.5 + 0.5; spark *= mix(0.4, 1.0, flicker); COLOR = teinte_halogene * spark * intensite_base; }

📄 shimmer_murs.gdshader

shader_type canvas_item; // Shimmer du liseré de mur (V5.8) — Candela 2D // Fait scintiller les arêtes et bordures de murs exposées à la torche. uniform float intensite_shimmer : hint_range(0.0, 2.0) = 0.8; uniform float frequence_scintillement : hint_range(0.5, 10.0) = 4.0; uniform vec4 couleur_lisere : source_color = vec4(0.98, 0.91, 0.80, 1.0); // Halogène uniform vec4 couleur_reflet : source_color = vec4(0.96, 0.69, 0.24, 1.0); // Ambre float noise(vec2 p) { vec2 i = floor(p); vec2 f = fract(p); f = f * f * (3.0 - 2.0 * f); float n = dot(i, vec2(1.0, 57.0)); return mix( mix(fract(sin(n + 0.0) * 43758.5453), fract(sin(n + 1.0) * 43758.5453), f.x), mix(fract(sin(n + 57.0) * 43758.5453), fract(sin(n + 58.0) * 43758.5453), f.x), f.y ); } void fragment() { vec4 base_color = texture(TEXTURE, UV); // Échantillonnage du bruit de micro-aspérités sur les arêtes vec2 world_pos = UV * 120.0; float n = noise(world_pos + vec2(TIME * frequence_scintillement * 0.2, 0.0)); // Pics de brillance spéculaire float glint = pow(n, 4.0) * intensite_shimmer; // Scintillement dynamique modulé float pulse = sin(TIME * frequence_scintillement + world_pos.x * 2.0) * 0.5 + 0.5; glint *= mix(0.5, 1.5, pulse); vec4 final_glow = mix(couleur_lisere, couleur_reflet, glint * 0.5) * glint; COLOR = base_color + final_glow * base_color.a; }
